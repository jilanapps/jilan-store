rules_version = '2';
service cloud.firestore {
  match /databases/{db}/documents {
    function isAdmin() {
      return request.auth != null
        && exists(/databases/$(db)/documents/masters/$(request.auth.uid));
    }
    function shopData(shopId) {
      return get(/databases/$(db)/documents/shops/$(shopId)).data;
    }
    function isShopSide(shopId) {
      let s = shopData(shopId);
      return request.auth != null
        && (s.retailerUid == request.auth.uid
            || s.distributorUid == request.auth.uid);
    }
    function ownsShop(shopId) {
      return isAdmin() || isShopSide(shopId);
    }
    match /masters/{uid} {
      allow read: if request.auth != null
        && (request.auth.uid == uid || isAdmin());
      allow write: if isAdmin();
    }
    match /users/{uid} {
      allow read: if request.auth != null
        && (request.auth.uid == uid || isAdmin());
      allow write: if request.auth != null
        && (request.auth.uid == uid || isAdmin());
    }
    match /shops/{shopId} {
      allow read: if request.auth != null && (
        resource.data.retailerUid == request.auth.uid ||
        resource.data.distributorUid == request.auth.uid || isAdmin());
      allow create: if request.auth != null && (
        request.resource.data.retailerUid == request.auth.uid || isAdmin());
      allow update: if request.auth != null && (
        resource.data.retailerUid == request.auth.uid ||
        resource.data.distributorUid == request.auth.uid || isAdmin());
      allow delete: if isAdmin();
      match /phones/{phoneId} {
        allow read, write: if ownsShop(shopId);
      }
      match /commands/{cmdId} {
        allow read: if ownsShop(shopId);
        allow create: if ownsShop(shopId)
          && request.resource.data.createdBy == request.auth.uid
          && request.resource.data.type in [
            'LOCK','UNLOCK','RELEASE','PAYMENT_RECEIVED','SET_DUE_DATE',
            'SET_PLAN','LOCATE','REMIND','UPDATE_NOW','FRP_ON','FRP_OFF','SET_MASTER_HASH'];
        allow update, delete: if ownsShop(shopId);
      }
      match /payments/{payId} {
        allow read, write: if ownsShop(shopId);
      }
      match /events/{evId} {
        allow read, write: if ownsShop(shopId);
      }
      match /paymentRequests/{requestId} {
        allow read: if ownsShop(shopId);
        allow create: if ownsShop(shopId)
          && request.resource.data.shopId == shopId
          && request.resource.data.tid is string
          && request.resource.data.tid.size() > 0
          && request.resource.data.amount is number
          && request.resource.data.amount > 0
          && request.resource.data.method in ['JazzCash', 'Easypaisa']
          && request.resource.data.status == 'pending'
          && request.resource.data.source == 'customer-tid'
          && request.resource.data.phoneDocId is string
          && request.resource.data.phoneDocId.size() > 0
          && !('dueDate' in request.resource.data)
          && !('installmentsLeft' in request.resource.data)
          && !('remainingAmount' in request.resource.data)
          && !('paidSoFar' in request.resource.data)
          && !('paymentsTotal' in request.resource.data)
          && !('planMonths' in request.resource.data)
          && !('monthlyInstallment' in request.resource.data)
          && !('price' in request.resource.data)
          && !('advance' in request.resource.data)
          && !('markupPercent' in request.resource.data);
        allow update: if ownsShop(shopId)
          && request.resource.data.tid == resource.data.tid
          && request.resource.data.amount == resource.data.amount
          && request.resource.data.method == resource.data.method
          && request.resource.data.phoneDocId == resource.data.phoneDocId
          && resource.data.status == 'pending'
          && request.resource.data.status in ['verified', 'rejected']
          && ((request.resource.data.status == 'verified'
               && request.resource.data.verifiedBy == request.auth.uid)
              || (request.resource.data.status == 'rejected'
                  && request.resource.data.rejectedBy == request.auth.uid
                  && request.resource.data.rejectReason is string
                  && request.resource.data.rejectReason.size() > 0));
        allow delete: if false;
      }
      match /audit/{auditId} {
        allow read: if ownsShop(shopId);
        allow create: if (isAdmin()
              && request.resource.data.createdBy == request.auth.uid)
            || (ownsShop(shopId)
              && request.resource.data.action in ['TID_VERIFIED', 'TID_REJECTED']
              && request.resource.data.actorUid == request.auth.uid);
        allow update, delete: if false;
      }
    }
    match /alerts/{alertId} {
      allow create: if request.auth != null;
      allow read, update, delete: if isAdmin();
    }

    // v51 KEY ACTIVATION (NOT published - the USER publishes this in
    // the Firebase Console, exactly like v49 was published).
    // Everything above this block = v50 FINAL verbatim (live v49 +
    // jilanKeys ADMIN-ONLY). v51 ADDS one narrow shop-side exception
    // and changes nothing else:
    //  - Master (admin) keeps full get/list/create/update/delete.
    //  - A shop Auth user may GET only a key whose shopId is a shop
    //    they own: the shops doc named by shopId, with retailerUid == uid or
    //    distributorUid == uid (the existing isShopSide() check).
    //    ShopId convention (verified in the customer app's doCloudLink):
    //    shops doc id = "shop_" + retailerUid stripped of
    //    non-alphanumerics, lowercased. A key with no shopId (advance /
    //    unassigned, or pre-shopId legacy) is therefore NOT readable
    //    by any shop and can never be shop-bound until Master assigns
    //    a shopId via Edit / Assign Shop.
    //  - A shop Auth user may UPDATE only their own shop's key, only
    //    unused -> used, only setting boundImei once, with key / shop /
    //    contact / shopId unchanged and no other field changed
    //    (diff().affectedKeys() == {status, boundImei}). They may NEVER
    //    list, create, delete, or change a bound or revoked key.
    //  - The v50 DRAFT's any-auth GET + bind hole stays CLOSED: there
    //    is no any-auth access anywhere in this file.
    // Key doc id = the key code itself (JLN-XXXX-XXXX-XXXX).
    // Stored status vocabulary: unused / used / revoked.
    match /jilanKeys/{keyId} {
      allow get, list, create, update, delete: if isAdmin();
      allow get: if request.auth != null
        && resource.data.shopId is string
        && resource.data.shopId.size() > 0
        && isShopSide(resource.data.shopId);
      allow update: if request.auth != null
        && resource.data.shopId is string
        && resource.data.shopId.size() > 0
        && isShopSide(resource.data.shopId)
        && resource.data.status == 'unused'
        && resource.data.boundImei == ''
        && request.resource.data.status == 'used'
        && request.resource.data.boundImei is string
        && request.resource.data.boundImei.size() > 0
        && request.resource.data.key == resource.data.key
        && request.resource.data.key == keyId
        && request.resource.data.shop == resource.data.shop
        && request.resource.data.contact == resource.data.contact
        && request.resource.data.shopId == resource.data.shopId
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'boundImei']);
    }

  }
}
